Step 1: Radar Fibonacci (Modifica colori, spessore, gradi)

import turtle

schermo = turtle.Screen()
schermo.bgcolor("black")
t = turtle.Turtle()
t.speed(0)

# --- MODIFICA QUI: COLORI, SPESSORE E GRADI ---
colori = ["cyan", "magenta", "yellow", "lime"]
spessore = 3      # Prova 1, 5 o 10
gradi = 90        # Prova 91, 89, 120 o 60
passi = 15
# ----------------------------------------------

a = 1
b = 1
for i in range(passi):
    t.pencolor(colori[i % len(colori)])
    t.pensize(spessore)
    for _ in range(90):
        t.forward(a * 0.04)
        t.left(gradi / 90)
    prossimo = a + b
    a = b
    b = prossimo
turtle.done()

Step 2: La Galassia Aurea (137.5°)

import turtle

schermo = turtle.Screen()
schermo.bgcolor("black")
turtle.tracer(0)

t = turtle.Turtle()
t.hideturtle()
t.penup()

colori = ["#00FFCC", "#FF007F", "#FFFF00", "#00BFFF", "#7FFF00", "#FF8C00"]
angolo_aureo = 137.508

for i in range(450):
    t.pencolor(colori[i % len(colori)])
    t.dot(4 + (i * 0.015))
    t.forward(i * 0.45)
    t.left(angolo_aureo)

turtle.update()
turtle.done()

import turtle

schermo = turtle.Screen()
schermo.bgcolor("black")
t = turtle.Turtle()
t.speed(5)

# --- MODIFICA QUI LA PAROLA SEGRETA ---
messaggio = "MAMMA"
# --------------------------------------

alfabeto = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
colori = ["#00FF00", "#00FFFF", "#FF0055", "#FFFF00"]
a = 1
b = 1
testo_cifrato = ""

for i in range(len(messaggio)):
    lettera = messaggio[i]
    if lettera in alfabeto:
        posizione = alfabeto.index(lettera)
        nuova_pos = (posizione + a) % 26
        nuova_lettera = alfabeto[nuova_pos]
    else:
        nuova_lettera = lettera
    testo_cifrato = testo_cifrato + nuova_lettera
    t.pencolor(colori[i % 4])
    t.pensize(3)
    t.forward(a * 5)
    t.left(a * 35)
    prossimo = a + b
    a = b
    b = prossimo

t.hideturtle()
print("ORIGINALE: " + messaggio)
print("CIFRATO:   " + testo_cifrato)
